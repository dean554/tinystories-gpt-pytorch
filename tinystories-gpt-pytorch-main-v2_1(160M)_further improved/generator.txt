custom_prompt = "hi"
encoded_prompt = tokenizer.encode(custom_prompt)
context_tokens = encoded_prompt.ids
context = torch.tensor([context_tokens], dtype=torch.long, device=device)

# CHANGED: reach through .module since generate() isn't defined on DataParallel
raw_model = model.module if isinstance(model, nn.DataParallel) else model

# CHANGED: eval mode disables dropout so generation isn't corrupted by random noise
raw_model.eval()

# CHANGED: no_grad avoids building an autograd graph for 1000 forward passes we'll never backward() through
with torch.no_grad():
    # CHANGED: autocast matches the precision used throughout training/eval
    with torch.amp.autocast('cuda', dtype=torch.float16):
        generated_indices = raw_model.generate(context, max_new_tokens=100, temperature=0.8)[0].tolist()

print(tokenizer.decode(generated_indices))

# CHANGED: flip back to train mode in case training continues after this cell
raw_model.train()
